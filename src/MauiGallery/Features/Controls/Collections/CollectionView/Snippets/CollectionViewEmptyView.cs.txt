// Uses the CommunityToolkit.Mvvm source generators
public partial class PlaceSearchViewModel : ObservableObject
{
    [ObservableProperty]
    public partial string SearchText { get; set; } = "";

    [ObservableProperty]
    [NotifyPropertyChangedFor(nameof(Summary))]
    public partial IReadOnlyList<Place> Results { get; private set; } = Places.All;

    public string Summary => $"{Results.Count} places";

    partial void OnSearchTextChanged(string value) =>
        Results = [.. Places.All.Where(p => p.Name.Contains(value, StringComparison.OrdinalIgnoreCase))];
}

// The header, footer and empty view can be plain strings or any view
var collectionView = new CollectionView
{
    ItemTemplate = placeTemplate,
    Header = summaryLabel,
    Footer = "Can't find a place? Suggest one.",
    EmptyView = new VerticalStackLayout
    {
        Padding = new Thickness(24, 48),
        Spacing = 8,
        Children =
        {
            new Label { Text = "No places found", FontAttributes = FontAttributes.Bold, HorizontalOptions = LayoutOptions.Center },
            new Label { Text = "Try a different city or country.", HorizontalOptions = LayoutOptions.Center },
        },
    },
};
collectionView.SetBinding(ItemsView.ItemsSourceProperty, static (PlaceSearchViewModel vm) => vm.Results);
