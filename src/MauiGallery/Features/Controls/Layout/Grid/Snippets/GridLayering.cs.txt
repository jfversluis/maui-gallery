var background = new Border
{
    StrokeThickness = 0,
    StrokeShape = new RoundRectangle { CornerRadius = 12 },
    Background = new LinearGradientBrush(
        [new GradientStop(Color.FromArgb("#0EA5E9"), 0), new GradientStop(Color.FromArgb("#6366F1"), 1)],
        new Point(0, 0),
        new Point(1, 1)),
};

var text = new VerticalStackLayout
{
    Padding = 16,
    VerticalOptions = LayoutOptions.End,
    Children =
    {
        new Label { Text = "Weekend in Lisbon", TextColor = Colors.White, FontSize = 20, FontAttributes = FontAttributes.Bold },
        new Label { Text = "3 nights, from €420", TextColor = Colors.White },
    },
};

var badge = new Border
{
    Margin = 12,
    Padding = new Thickness(8, 2),
    HorizontalOptions = LayoutOptions.End,
    VerticalOptions = LayoutOptions.Start,
    BackgroundColor = Colors.White,
    StrokeThickness = 0,
    StrokeShape = new RoundRectangle { CornerRadius = 10 },
    Content = new Label { Text = "NEW", FontSize = 12, FontAttributes = FontAttributes.Bold, TextColor = Color.FromArgb("#4F46E5") },
};

// Children in the same cell are drawn in the order they're added.
var card = new Grid { HeightRequest = 160 };
card.Add(background);
card.Add(text);
card.Add(badge);
