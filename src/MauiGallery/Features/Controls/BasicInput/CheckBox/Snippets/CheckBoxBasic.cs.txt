var rememberMe = new CheckBox
{
    IsChecked = $(IsChecked),
    IsEnabled = $(IsEnabled),
};
SemanticProperties.SetDescription(rememberMe, "Remember me");
rememberMe.CheckedChanged += OnCheckedChanged;

var label = new Label { Text = "Remember me", VerticalOptions = LayoutOptions.Center };
var tap = new TapGestureRecognizer();
tap.Tapped += (_, _) =>
{
    if (rememberMe.IsEnabled)
    {
        rememberMe.IsChecked = !rememberMe.IsChecked;
    }
};
label.GestureRecognizers.Add(tap);

var row = new HorizontalStackLayout { Spacing = 4, Children = { rememberMe, label } };

void OnCheckedChanged(object? sender, CheckedChangedEventArgs e)
{
    bool isChecked = e.Value;
}
