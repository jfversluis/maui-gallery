var playButton = new ImageButton
{
    WidthRequest = 64,
    HeightRequest = 64,
    Padding = 16,
    BackgroundColor = $(BackgroundColor),
    CornerRadius = $(CornerRadius),

    Source = new FontImageSource

        FontFamily = "FluentIcons",

        Color = Colors.White,
    },
};

SemanticProperties.SetDescription(playButton, "Play");
