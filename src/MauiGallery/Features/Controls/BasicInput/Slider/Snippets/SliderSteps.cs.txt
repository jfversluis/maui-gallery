const double Step = $(Step);

void OnFontSizeChanged(object? sender, ValueChangedEventArgs e)
{
    // Snap to the nearest step. Setting Value raises ValueChanged once more with the snapped value.
    FontSizeSlider.Value = Math.Round(e.NewValue / Step) * Step;
}

{
    // The user grabbed the thumb
}

{
    // The user let go: a good moment to save the final value
}
