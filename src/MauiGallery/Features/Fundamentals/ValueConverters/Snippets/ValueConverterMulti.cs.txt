public class AllTrueConverter : IMultiValueConverter
{
    public object? Convert(object?[] values, Type targetType, object? parameter, CultureInfo culture) =>
        values.All(value => value is true);

    public object?[] ConvertBack(object? value, Type[] targetTypes, object? parameter, CultureInfo culture) =>
        throw new NotSupportedException();
}

button.SetBinding(Button.IsEnabledProperty, new MultiBinding
{
    Converter = new AllTrueConverter(),
    Bindings =
    {
        BindingBase.Create(static (CheckBox c) => c.IsChecked, source: terms),
        BindingBase.Create(static (CheckBox c) => c.IsChecked, source: age),
    },
});
