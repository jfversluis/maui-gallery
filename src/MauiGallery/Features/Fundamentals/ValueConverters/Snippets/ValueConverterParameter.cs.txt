public class PluralConverter : IValueConverter
{
    public object? Convert(object? value, Type targetType, object? parameter, CultureInfo culture)
    {
        var count = (int)Math.Round(System.Convert.ToDouble(value, culture));
        var noun = parameter as string ?? "item";
        return count switch
        {
            0 => $"No {noun}s",
            1 => $"1 {noun}",
            _ => $"{count} {noun}s",
        };
    }

    public object? ConvertBack(object? value, Type targetType, object? parameter, CultureInfo culture) =>
        throw new NotSupportedException();
}

label.SetBinding(
    Label.TextProperty,
    static (Stepper s) => s.Value,
    source: count,
    converter: new PluralConverter(),
    converterParameter: $(Noun));
