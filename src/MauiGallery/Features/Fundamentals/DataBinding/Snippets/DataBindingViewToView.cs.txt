var slider = new Slider { Maximum = 360, Value = 30 };
var label = new Label { FontSize = 32, HorizontalOptions = LayoutOptions.Center };

// Lambda bindings are compiled: no reflection, and the path is checked by the compiler
label.SetBinding(Label.TextProperty, static (Slider s) => s.Value, source: slider, stringFormat: "{0:F0}°");
label.SetBinding(Label.RotationProperty, static (Slider s) => s.Value, source: slider);
