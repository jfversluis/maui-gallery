// MauiProgram.cs: builder.Services.AddSingleton(DeviceInfo.Current);
// Without dependency injection, use DeviceInfo.Current directly.
public class DeviceInfoViewModel(IDeviceInfo deviceInfo)
{
    public string Model => deviceInfo.Model;
    public string Manufacturer => deviceInfo.Manufacturer;
    public string Name => deviceInfo.Name;
    public DevicePlatform Platform => deviceInfo.Platform;
    public string Version => deviceInfo.VersionString;
    public DeviceIdiom Idiom => deviceInfo.Idiom;

    // Physical or Virtual (a simulator or emulator)
    public DeviceType DeviceType => deviceInfo.DeviceType;

    public bool IsPhone => deviceInfo.Idiom == DeviceIdiom.Phone;
}
