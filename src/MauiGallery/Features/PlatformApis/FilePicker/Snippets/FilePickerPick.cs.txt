// MauiProgram.cs: builder.Services.AddSingleton(FilePicker.Default);
// Without dependency injection, use FilePicker.Default directly.
public partial class FilePickerViewModel(IFilePicker filePicker) : ObservableObject
{
    // Name the file types per platform
    private static readonly FilePickerFileType TextFiles = new(new Dictionary<DevicePlatform, IEnumerable<string>>
    {
        [DevicePlatform.iOS] = ["public.plain-text"],
        [DevicePlatform.MacCatalyst] = ["public.plain-text"],
        [DevicePlatform.Android] = ["text/plain"],
        [DevicePlatform.WinUI] = [".txt"],
    });

    [ObservableProperty]
    public partial string? FileName { get; set; }

    [ObservableProperty]
    public partial string? TextPreview { get; set; }

    [ObservableProperty]
    public partial ImageSource? ImagePreview { get; set; }

    [RelayCommand]
    private async Task PickAsync()
    {
        FileResult? file = await filePicker.PickAsync(new PickOptions
        {
            PickerTitle = "Choose a file",
            FileTypes = $(FileTypes),
        });
        if (file is null)
        {
            return; // Cancelled
        }

        FileName = file.FileName;

        // The picked file may be a temporary copy: read it now, or copy it to AppDataDirectory.
        // It can be any size, so read only what you need.
        await using var stream = await file.OpenReadAsync();

        if (file.ContentType.StartsWith("image/"))
        {
            // Decoding a huge image can run out of memory. The image opens the file itself.
            if (stream.Length <= 20 * 1024 * 1024)
            {
                ImagePreview = ImageSource.FromStream(_ => file.OpenReadAsync());
            }
        }
        else if (file.ContentType.StartsWith("text/"))
        {
            using var reader = new StreamReader(stream);
            var buffer = new char[400];
            var read = await reader.ReadBlockAsync(buffer);
            TextPreview = new string(buffer, 0, read);
        }
    }
}
