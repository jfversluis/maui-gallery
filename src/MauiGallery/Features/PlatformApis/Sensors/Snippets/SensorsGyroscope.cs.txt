// IsMonitoring is app-wide: another page or window may already be using the sensor
if (Gyroscope.Default.IsSupported && !Gyroscope.Default.IsMonitoring)
{
    Gyroscope.Default.Start(SensorSpeed.$(Speed));
    Gyroscope.Default.ReadingChanged += OnGyroscopeReadingChanged;
}

// Stop when you're done, at the latest when the page disappears
Gyroscope.Default.Stop();
Gyroscope.Default.ReadingChanged -= OnGyroscopeReadingChanged;

private void OnGyroscopeReadingChanged(object? sender, GyroscopeChangedEventArgs e)
{
    // Radians per second around each axis
    Vector3 angularVelocity = e.Reading.AngularVelocity;
}
