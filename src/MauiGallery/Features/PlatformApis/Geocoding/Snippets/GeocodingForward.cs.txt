if (Connectivity.Current.NetworkAccess != NetworkAccess.Internet)
{
    // Geocoding asks an online service
    return;
}

var best = locations.FirstOrDefault();
if (best is not null)
{
    Debug.WriteLine($"{best.Latitude}, {best.Longitude}");
}
