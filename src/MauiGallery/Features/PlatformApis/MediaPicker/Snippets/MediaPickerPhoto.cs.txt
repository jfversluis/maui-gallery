List<FileResult> photos = await MediaPicker.Default.PickPhotosAsync(new MediaPickerOptions
{
    Title = "Choose a photo",
    SelectionLimit = 1,
    MaximumWidth = $(Size),
    MaximumHeight = $(Size),
    CompressionQuality = $(Quality),
});

var photo = photos.FirstOrDefault();
if (photo is null)
{
    return; // Cancelled
}

// Show it straight from the picker's file, which can be temporary.
// To keep the photo, copy it to FileSystem.AppDataDirectory.
PhotoImage.Source = ImageSource.FromStream(_ => photo.OpenReadAsync());
