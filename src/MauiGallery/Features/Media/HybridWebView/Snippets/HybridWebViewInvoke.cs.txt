// The page calls these methods with window.HybridWebView.InvokeDotNet
public sealed class HybridBridge(IDeviceInfo deviceInfo)
{
    public string Greet(string name) => $"Hello {name}, from .NET!";

    public DeviceSummary GetDeviceInfo() =>
        new(deviceInfo.Platform.ToString(), deviceInfo.VersionString, deviceInfo.Model, deviceInfo.Idiom.ToString());
}

public sealed record DeviceSummary(string Platform, string Version, string Model, string Idiom);

// Source-generated JSON for every type that crosses the boundary: trim-safe, no reflection
[JsonSourceGenerationOptions(PropertyNamingPolicy = JsonKnownNamingPolicy.CamelCase)]
[JsonSerializable(typeof(DeviceSummary))]
[JsonSerializable(typeof(string))]
internal sealed partial class HybridJsonContext : JsonSerializerContext;

// In the page's constructor. A source generator replaces this call with reflection-free code, but it can't see
// x:Name fields generated from XAML: call it through a typed local, or it throws at run time.
HybridWebView webView = WebPage;
webView.SetInvokeJavaScriptTarget(new HybridBridge(DeviceInfo.Current), HybridJsonContext.Default);
