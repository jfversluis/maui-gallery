using Microsoft.Maui.Controls.Maps;
using Microsoft.Maui.Maps;
using Map = Microsoft.Maui.Controls.Maps.Map;

var map = new Map
{
    HeightRequest = 360,
    MapType = MapType.$(MapType),
    IsScrollEnabled = $(Scroll),
    IsZoomEnabled = $(Zoom),
    IsTrafficEnabled = $(Traffic),
};

var amsterdam = new Location(52.3676, 4.9041);
map.MoveToRegion(MapSpan.FromCenterAndRadius(amsterdam, Distance.FromKilometers(3)));
