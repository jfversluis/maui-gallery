var page = new NotesPage();
var window = new Window(page)
{
    Title = $(Title),
    MinimumWidth = 360,
    MinimumHeight = 360,
};

// The page binds to its window's size and position
page.BindingContext = window;

// Mac Catalyst ignores a frame set before the window is on screen, so set it on first activation
window.Activated += OnFirstActivated;
Application.Current?.OpenWindow(window);

void OnFirstActivated(object? sender, EventArgs e)
{
    window.Activated -= OnFirstActivated;
    window.X = $(X);
    window.Y = $(Y);
    window.Width = $(Width);
    window.Height = $(Height);
}

// Later: close it
Application.Current?.CloseWindow(window);
