// App.xaml.cs
protected override Window CreateWindow(IActivationState? activationState)
{
    var shell = new AppShell();
    var window = new Window(shell);

    // Shell is never measured, so follow the window's size instead
    window.SizeChanged += (_, _) => shell.ApplyLayout(window.Width, window.Height);
    return window;
}

// AppShell.xaml.cs
// Follow the window, not the device: a foldable opens up, an iPad app shares the screen
public void ApplyLayout(double windowWidth, double windowHeight)
{
    var isDesktop = DeviceInfo.Platform == DevicePlatform.WinUI || DeviceInfo.Platform == DevicePlatform.MacCatalyst;
    var useTabs = !isDesktop && (windowWidth < 640 || windowHeight < 480);
    FlyoutBehavior = useTabs ? FlyoutBehavior.Disabled
        : windowWidth >= 1100 && windowHeight >= 480 ? FlyoutBehavior.Locked
        : FlyoutBehavior.Flyout;

    foreach (var tab in Items.SelectMany(item => item.Items))
    {
        Shell.SetTabBarIsVisible(tab, useTabs);

        // Shell copies the value onto each ShellContent and open page and never updates it. Android reads the
        // ShellContent first, so clear it there and update the open pages
        foreach (var content in tab.Items)
        {
            content.ClearValue(Shell.TabBarIsVisibleProperty);
        }

        var pages = tab.Items.Select(content => ((IShellContentController)content).Page)
            .Concat(tab.Navigation.NavigationStack)
            .OfType<Page>();
        foreach (var page in pages)
        {
            Shell.SetTabBarIsVisible(page, useTabs);
        }
    }
}
